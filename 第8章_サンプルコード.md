## 8.5 挨拶アプリ（HTML）

```html
<!DOCTYPE html>
<html lang="ja">
  <head>
    <meta charset="UTF-8" />
    <title>挨拶アプリ</title>
  </head>
  <body>
    <h1>挨拶アプリ</h1>
    <input type="text" id="nameInput" placeholder="名前を入力" />
    <button id="greetButton">挨拶する</button>
    <p id="result"></p>

    <script>
      // ここにJavaScriptを書く
      const button = document.querySelector("#greetButton");

      button.addEventListener("click", function () {
        // 1. 入力欄の要素を取得
        const input = document.querySelector("#nameInput");
        // 2. 結果を表示する場所を取得
        const result = document.querySelector("#result");

        // 3. 入力された値（value）を取り出す
        const name = input.value;

        // 4. 挨拶メッセージを作って表示
        if (name === "") {
          result.textContent = "名前を入力してください！";
        } else {
          result.textContent = "こんにちは、" + name + "さん！";
        }
      });
    </script>
  </body>
</html>
```

## 8.6 querySelectorAll

```js
// すべてのボタンを取得
const buttons = document.querySelectorAll("button");

// すべての .item クラス要素を取得
const items = document.querySelectorAll(".item");

// 取得した要素の数を確認
console.log("ボタンの数: " + buttons.length);

// forEachで全要素に処理を適用
items.forEach(function (item) {
  item.style.color = "blue";
});
```

## 8.9 ToDo リストアプリ

```html
<!DOCTYPE html>
<html lang="ja">
  <head>
    <meta charset="UTF-8" />
    <title>ToDoリスト</title>
    <style>
      .todo-item {
        padding: 8px;
        margin: 4px 0;
        background: #f0f0f0;
        border-radius: 4px;
        display: flex;
        justify-content: space-between;
        align-items: center;
      }
      .delete-btn {
        background: #ff4444;
        color: white;
        border: none;
        padding: 4px 8px;
        border-radius: 4px;
        cursor: pointer;
      }
    </style>
  </head>
  <body>
    <h1>ToDoリスト</h1>
    <div>
      <input type="text" id="todoInput" placeholder="やることを入力" />
      <button id="addButton">追加</button>
    </div>
    <div id="todoList"></div>

    <script>
      // 要素の取得
      const input = document.querySelector("#todoInput");
      const addButton = document.querySelector("#addButton");
      const todoList = document.querySelector("#todoList");

      // 追加ボタンがクリックされたとき
      addButton.addEventListener("click", function () {
        const todoText = input.value;

        // 入力が空なら何もしない
        if (todoText === "") {
          alert("内容を入力してください");
          return;
        }

        // 新しいToDoアイテムを作成
        const todoItem = document.createElement("div");
        todoItem.className = "todo-item";

        // テキスト部分
        const textSpan = document.createElement("span");
        textSpan.textContent = todoText;

        // 削除ボタン
        const deleteButton = document.createElement("button");
        deleteButton.textContent = "削除";
        deleteButton.className = "delete-btn";

        // 削除ボタンがクリックされたとき
        deleteButton.addEventListener("click", function () {
          todoItem.remove();
        });

        // 要素を組み立てる
        todoItem.appendChild(textSpan);
        todoItem.appendChild(deleteButton);

        // リストに追加
        todoList.appendChild(todoItem);

        // 入力欄をクリア
        input.value = "";
      });

      // Enterキーでも追加できるようにする
      input.addEventListener("keypress", function (event) {
        if (event.key === "Enter") {
          addButton.click();
        }
      });
    </script>
  </body>
</html>
```

## 8.10 様々なイベント

```js
const input = document.querySelector("#myInput");

// 入力中にリアルタイムで反応
input.addEventListener("input", function () {
  console.log("現在の値: " + input.value);
});

// 入力が確定したとき
input.addEventListener("change", function () {
  console.log("確定した値: " + input.value);
});

// Enterキーが押されたとき
input.addEventListener("keypress", function (event) {
  if (event.key === "Enter") {
    console.log("Enterが押されました");
  }
});
```

## 8.11 イベントオブジェクト

```js
button.addEventListener("click", function (event) {
  // クリックされた要素
  console.log(event.target);

  // マウスの座標
  console.log("X座標: " + event.clientX);
  console.log("Y座標: " + event.clientY);
});

input.addEventListener("keypress", function (event) {
  // 押されたキー
  console.log("押されたキー: " + event.key);

  // デフォルトの動作を防ぐ
  if (event.key === "Enter") {
    event.preventDefault();
  }
});
```
