## 12.2 script.js

```javascript
// 要素を取得する
const itemInput = document.getElementById("item-input");
const addButton = document.getElementById("add-button");
const inventoryList = document.getElementById("inventory-list");

// 追加ボタンをクリックしたときの処理
addButton.addEventListener("click", function () {
  const text = itemInput.value; // 入力された文字を取得

  // 空欄の場合は何もしない
  if (text === "") {
    return;
  }

  // 新しいリストアイテム(li)を作成
  const newItem = document.createElement("li");
  newItem.textContent = text;

  // リストに追加
  inventoryList.appendChild(newItem);

  // 入力欄を空にする
  itemInput.value = "";
});
```

## 12.5 script.js (完成版)

```javascript
const itemInput = document.getElementById("item-input");
const addButton = document.getElementById("add-button");
const inventoryList = document.getElementById("inventory-list");

// ページ読み込み時にデータを復元
document.addEventListener("DOMContentLoaded", loadItems);

addButton.addEventListener("click", addItem);

function addItem() {
  const text = itemInput.value;
  if (text === "") return;

  createItemElement(text);
  saveItems(); // データを保存

  itemInput.value = "";
}

function createItemElement(text) {
  const newItem = document.createElement("li");

  const span = document.createElement("span");
  span.textContent = text;
  newItem.appendChild(span);

  const deleteButton = document.createElement("button");
  deleteButton.textContent = "削除";
  deleteButton.addEventListener("click", function () {
    inventoryList.removeChild(newItem);
    saveItems(); // 削除した後も保存
  });

  newItem.appendChild(deleteButton);
  inventoryList.appendChild(newItem);
}

function saveItems() {
  const items = [];
  const listItems = inventoryList.querySelectorAll("li span");

  listItems.forEach((span) => {
    items.push(span.textContent);
  });

  // 配列をJSON文字列に変換して保存
  localStorage.setItem("inventory", JSON.stringify(items));
}

function loadItems() {
  const itemsJSON = localStorage.getItem("inventory");
  if (itemsJSON) {
    const items = JSON.parse(itemsJSON);
    items.forEach((text) => {
      createItemElement(text);
    });
  }
}
```

## 12.7 script.js (完全版)

```javascript
const itemInput = document.getElementById("item-input");
const quantityInput = document.getElementById("quantity-input");
const addButton = document.getElementById("add-button");
const inventoryList = document.getElementById("inventory-list");

let inventory = []; // データを配列で管理

document.addEventListener("DOMContentLoaded", loadItems);
addButton.addEventListener("click", addItem);

function addItem() {
  const name = itemInput.value.trim();
  const quantity = parseInt(quantityInput.value);

  if (name === "" || isNaN(quantity) || quantity < 1) {
    alert("正しい値を入力してください");
    return;
  }

  // 既に同じ名前のアイテムがあれば個数を加算
  const existingItem = inventory.find((item) => item.name === name);
  if (existingItem) {
    existingItem.quantity += quantity;
  } else {
    // 新しいアイテムを追加
    inventory.push({
      id: Date.now(), // 一意のIDを生成
      name: name,
      quantity: quantity,
    });
  }

  saveAndRender();
  itemInput.value = "";
  quantityInput.value = "1";
}

function deleteItem(id) {
  inventory = inventory.filter((item) => item.id !== id);
  saveAndRender();
}

function updateQuantity(id, newQuantity) {
  const item = inventory.find((item) => item.id === id);
  if (item) {
    item.quantity = newQuantity;
    if (item.quantity <= 0) {
      deleteItem(id);
    } else {
      saveAndRender();
    }
  }
}

function renderItems() {
  inventoryList.innerHTML = ""; // リストをクリア

  inventory.forEach((item) => {
    const li = document.createElement("li");

    // アイテム名
    const nameSpan = document.createElement("span");
    nameSpan.className = "item-name";
    nameSpan.textContent = item.name;

    // 個数コントロール
    const quantityControl = document.createElement("div");
    quantityControl.className = "quantity-control";

    const decreaseBtn = document.createElement("button");
    decreaseBtn.textContent = "-";
    decreaseBtn.className = "quantity-btn";
    decreaseBtn.addEventListener("click", () => {
      updateQuantity(item.id, item.quantity - 1);
    });

    const quantitySpan = document.createElement("span");
    quantitySpan.className = "quantity";
    quantitySpan.textContent = item.quantity;

    const increaseBtn = document.createElement("button");
    increaseBtn.textContent = "+";
    increaseBtn.className = "quantity-btn";
    increaseBtn.addEventListener("click", () => {
      updateQuantity(item.id, item.quantity + 1);
    });

    quantityControl.appendChild(decreaseBtn);
    quantityControl.appendChild(quantitySpan);
    quantityControl.appendChild(increaseBtn);

    // 削除ボタン
    const deleteBtn = document.createElement("button");
    deleteBtn.textContent = "削除";
    deleteBtn.className = "delete-btn";
    deleteBtn.addEventListener("click", () => {
      if (confirm(`「${item.name}」を削除しますか？`)) {
        deleteItem(item.id);
      }
    });

    li.appendChild(nameSpan);
    li.appendChild(quantityControl);
    li.appendChild(deleteBtn);
    inventoryList.appendChild(li);
  });
}

function saveAndRender() {
  localStorage.setItem("inventory", JSON.stringify(inventory));
  renderItems();
}

function loadItems() {
  const saved = localStorage.getItem("inventory");
  if (saved) {
    inventory = JSON.parse(saved);
    renderItems();
  }
}
```
