# 第 10 章 サンプルコード

## 10.1 データの保存

```js
// localStorage.setItem("キー", "値");
localStorage.setItem("userName", "先生");
```

## 10.2 データの読み込み

```js
// const 値 = localStorage.getItem("キー");
const name = localStorage.getItem("userName");
console.log(name); // "先生"
```

## 10.3 データの削除

```js
// localStorage.removeItem("キー");
localStorage.removeItem("userName");
```

## 10.4 JSON への変換

```js
const item = {
  id: 1,
  name: "財務用電卓",
  count: 5,
};

// オブジェクトをJSON文字列に変換
const jsonString = JSON.stringify(item);
console.log(jsonString);
// 結果: '{"id":1,"name":"財務用電卓","count":5}'

// ローカルストレージに保存
localStorage.setItem("inventoryData", jsonString);
```

## 10.5 JSON からの復元

```js
// ローカルストレージから読み込み
const savedData = localStorage.getItem("inventoryData");

if (savedData) {
  // JSON文字列をオブジェクトに戻す
  const item = JSON.parse(savedData);
  console.log(item.name); // "財務用電卓"
}
```

## 10.6 保存機能付きメモ帳

```html
<!DOCTYPE html>
<html lang="ja">
  <head>
    <meta charset="UTF-8" />
    <title>保存できるメモ</title>
  </head>
  <body>
    <h1>保存できるメモ</h1>
    <textarea id="memoArea" rows="5" cols="30"></textarea>
    <br />
    <button id="saveButton">保存する</button>
    <button id="clearButton">削除する</button>

    <script>
      const memoArea = document.querySelector("#memoArea");
      const saveButton = document.querySelector("#saveButton");
      const clearButton = document.querySelector("#clearButton");

      // 1. ページ読み込み時にデータを復元
      const savedMemo = localStorage.getItem("myMemo");
      if (savedMemo) {
        memoArea.value = savedMemo;
      }

      // 2. 保存ボタンの処理
      saveButton.addEventListener("click", function () {
        const text = memoArea.value;
        localStorage.setItem("myMemo", text);
        alert("保存しました！");
      });

      // 3. 削除ボタンの処理
      clearButton.addEventListener("click", function () {
        localStorage.removeItem("myMemo");
        memoArea.value = ""; // 画面もクリア
        alert("削除しました！");
      });
    </script>
  </body>
</html>
```
